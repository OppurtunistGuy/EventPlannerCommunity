   316→   316→  const cartTotal = cart.reduce((sum, c) => sum + c.menuItem.price * c.quantity, 0);

   318→   318→
   319→   319→  const removeFromCart = (itemId: string) => {
   320→   320→    setCart(cart.filter(c => c.menuItem.id !== itemId));
   321→   321→  };
   322→   322→
   323→   323→  const updateCartQty = (itemId: string, delta: number) => {

   325→   325→      if (c.menuItem.id !== itemId) return c;

   413→   413→export function BillModal() {
   414→   414→  const {
   415→   415→    showBill, setShowBill, billData, setBillData,
   416→   416→    selectedTable, endSession,
   417→   417→  } = useApp();
   418→   418→
   419→   419→  const settleBill = async () => {
   420→   420→    if (!billData || !selectedTable) return;
   421→   421→    try {
   422→   422→      const billId = (billData as Record<string, unknown>).billId as string;
   423→   423→      if (billId) {
   424→   424→        await fetch('/api/bill', {
   425→   425→          method: 'PATCH',
   426→   426→          headers: { 'Content-Type': 'application/json' },
   427→   427→          body: JSON.stringify({ billId }),
   428→   428→        });
   429→   429→      } else {
   430→   430→        const orders = (billData as Record<string, unknown>).orders as { id: string }[];
   431→   431→        for (const order of orders) {
   432→   432→          await fetch('/api/orders', {
   433→   433→            method: 'PATCH',
   434→   434→            headers: { 'Content-Type': 'application/json' },
   435→   435→            body: JSON.stringify({ orderId: order.id, status: 'billed' }),
   436→   436→          });
   437→   437→        }
   438→   438→      }
   439→   439→      setShowBill(false);
   440→   440→      setBillData(null);

   443→   443→      console.error(err);
   444→   444→    }

   447→   447→  return (
   448→   448→    <AnimatePresence>

   460→   460→                <span className="px-3 py-1 bg-white/15 rounded-full text-sm font-medium">Table {(billData as Record<string, unknown>).tableNumber}</span>
