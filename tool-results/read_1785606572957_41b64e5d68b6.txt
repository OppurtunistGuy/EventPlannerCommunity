   233→   233→export function LookupModal() {
   234→   234→  const {
   235→   235→    showLookupModal, setShowLookupModal,
   236→   236→    lookupCode, setLookupCode,
   237→   237→    lookupLoading, lookupError, setLookupError,
   238→   238→    lookupReservationByCode,
   239→   239→  } = useApp();
   240→   240→
   241→   241→  const inputRef = useRef<HTMLInputElement>(null);
   242→   242→
   243→   243→  const handleLookup = async (code?: string) => {
   244→   244→    const codeToUse = code || lookupCode;
   245→   245→    if (!codeToUse.trim() || codeToUse.trim().length !== 6) return;
   246→   246→    const success = await lookupReservationByCode(codeToUse);
   247→   247→    if (success) {
   248→   248→      setShowLookupModal(false);
   249→   249→    }
   250→   250→  };
   251→   251→
   252→   252→  const handleCodeChange = (value: string) => {
   253→   253→    // Only allow digits
   254→   254→    const digits = value.replace(/\D/g, '').slice(0, 6);
   255→   255→    setLookupCode(digits);
   256→   256→    setLookupError('');
   257→   257→    // Auto-submit when 6 digits are entered
   258→   258→    if (digits.length === 6) {
   259→   259→      setTimeout(() => handleLookup(digits), 100);
   260→   260→    }
   261→   261→  };
   262→   262→
   263→   263→  // Focus input when modal opens
   264→   264→  useEffect(() => {
   265→   265→    if (showLookupModal) {
   266→   266→      setTimeout(() => inputRef.current?.focus(), 200);
   267→   267→    }
   268→   268→  }, [showLookupModal]);
   269→   269→
   270→   270→  return (
   271→   271→    <AnimatePresence>
   272→   272→      {showLookupModal && (
   273→   273→        <motion.div initial={{ opacity: 0 }} animate={{ opacity: 1 }} exit={{ opacity: 0 }}
   274→   274→          className="fixed inset-0 z-50 bg-black/50 flex items-center justify-center p-4"
   275→   275→          onClick={() => { setShowLookupModal(false); setLookupError(''); }}>
   276→   276→          <motion.div variants={scaleIn} initial="hidden" animate="visible" exit="exit"
   277→   277→            onClick={e => e.stopPropagation()}
   278→   278→            className="bg-white rounded-2xl w-full max-w-md shadow-2xl">
   279→   279→            <div className="p-6">
   280→   280→              <div className="flex items-center justify-between mb-6">
   281→   281→                <div>
   282→   282→                  <h2 className="font-semibold text-lg">My Reservation</h2>
   283→   283→                  <p className="text-xs text-[var(--color-muted-foreground)] mt-0.5">Enter your reservation code to view your table and order</p>
   284→   284→                </div>
   285→   285→                <button onClick={() => { setShowLookupModal(false); setLookupError(''); }} className="p-1.5 hover:bg-[var(--color-secondary)] rounded-lg transition-colors">
   286→   286→                  <X className="w-5 h-5" />
   287→   287→                </button>
   288→   288→              </div>
   289→   289→              <div className="space-y-4">
   290→   290→                <div className="bg-[var(--color-secondary)] rounded-xl p-4 flex items-center gap-3">
   291→   291→                  <div className="w-10 h-10 rounded-full bg-[var(--color-accent)]/10 flex items-center justify-center flex-shrink-0">
   292→   292→                    <Ticket className="w-5 h-5 text-[var(--color-accent)]" />
   293→   293→                  </div>
   294→   294→                  <div>
   295→   295→                    <p className="text-sm font-medium">Have a reservation code?</p>
   296→   296→                    <p className="text-xs text-[var(--color-muted-foreground)]">Enter the 6-digit code you received when you reserved</p>
   297→   297→                  </div>
   298→   298→                </div>
   299→   299→                <div>
   300→   300→                  <label className="block text-xs font-medium text-[var(--color-muted-foreground)] mb-1.5">Reservation Code</label>
   301→   301→                  <input
   302→   302→                    ref={inputRef}
   303→   303→                    type="text"
   304→   304→                    inputMode="numeric"
   305→   305→                    pattern="\d{6}"
   306→   306→                    value={lookupCode}
   307→   307→                    onChange={e => handleCodeChange(e.target.value)}
   308→   308→                    onPaste={e => {
   309→   309→                      const pasted = e.clipboardData.getData('text').replace(/\D/g, '').slice(0, 6);
   310→   310→                      if (pasted.length === 6) {
   311→   311→                        setLookupCode(pasted);
   312→   312→                        setTimeout(() => handleLookup(pasted), 100);
   313→   313→                      } else {
   314→   314→                        setLookupCode(pasted);
   315→   315→                      }
   316→   316→                    }}
   317→   317→                    placeholder="Enter 6-digit code"
   318→   318→                    maxLength={6}
   319→   319→                    className="w-full px-4 py-3 border border-[var(--color-border)] rounded-xl text-lg font-mono text-center tracking-[0.5em] focus:outline-none focus:ring-2 focus:ring-[var(--color-primary)]/20 focus:border-[var(--color-primary)]"
   320→   320→                    onKeyDown={e => { if (e.key === 'Enter') handleLookup(); }}
   321→   321→                  />
   322→   322→                  <div className="flex items-center justify-center gap-1 mt-2">
   323→   323→                    {[0,1,2,3,4,5].map(i => (
   324→   324→                      <div key={i} className={`w-2 h-2 rounded-full transition-colors ${i < lookupCode.length ? 'bg-[var(--color-primary)]' : 'bg-[var(--color-border)]'}`} />
   325→   325→                    ))}
   326→   326→                  </div>
   327→   327→                </div>
   328→   328→                <div className="flex items-center gap-2 justify-center">
   329→   329→                  <span className="text-xs text-[var(--color-muted-foreground)] bg-[var(--color-secondary)] px-2.5 py-1 rounded-full">Test code: 777777 (Table 7)</span>
   330→   330→                </div>
   331→   331→                {lookupError && (
   332→   332→                  <div className="bg-[var(--color-destructive)]/10 border border-[var(--color-destructive)]/20 rounded-xl p-3 text-center">
   333→   333→                    <p className="text-sm text-[var(--color-destructive)]">{lookupError}</p>
   334→   334→                  </div>
   335→   335→                )}
   336→   336→                <button
   337→   337→                  onClick={() => handleLookup()}
   338→   338→                  disabled={lookupLoading || lookupCode.length !== 6}
   339→   339→                  className="w-full py-3 bg-[var(--color-accent)] text-white rounded-xl font-semibold text-sm hover:opacity-90 transition-opacity disabled:opacity-50 flex items-center justify-center gap-2"
   340→   340→                >
   341→   341→                  {lookupLoading ? <RefreshCw className="w-4 h-4 animate-spin" /> : <Key className="w-4 h-4" />}
   342→   342→                  {lookupLoading ? 'Looking up...' : 'Find My Reservation'}
   343→   343→                </button>
   344→   344→              </div>
   345→   345→            </div>
   346→   346→          </motion.div>
   347→   347→        </motion.div>
   348→   348→      )}
   349→   349→    </AnimatePresence>
   350→   350→  );
   351→   351→}
