   237→// ==========================================
   238→// LOOKUP RESERVATION MODAL
   239→// ==========================================
   240→export function LookupModal() {
   241→  const {
   242→    showLookupModal, setShowLookupModal,
   243→    lookupCode, setLookupCode,
   244→    lookupLoading, lookupError, setLookupError,
   245→    lookupReservationByCode,
   246→  } = useApp();
   247→
   248→  const inputRef = useRef<HTMLInputElement>(null);
   249→
   250→  const handleLookup = async (code?: string) => {
   251→    const codeToUse = code || lookupCode;
   252→    if (!codeToUse.trim() || codeToUse.trim().length !== 6) return;
   253→    const success = await lookupReservationByCode(codeToUse);
   254→    if (success) {
   255→      setShowLookupModal(false);
   256→    }
   257→  };
   258→
   259→  const handleCodeChange = (value: string) => {
   260→    // Only allow digits
   261→    const digits = value.replace(/\D/g, '').slice(0, 6);
   262→    setLookupCode(digits);
   263→    setLookupError('');
   264→    // Auto-submit when 6 digits are entered
   265→    if (digits.length === 6) {
   266→      setTimeout(() => handleLookup(digits), 100);
   267→    }
   268→  };
   269→
   270→  // Focus input when modal opens
   271→  useEffect(() => {
   272→    if (showLookupModal) {
   273→      setTimeout(() => inputRef.current?.focus(), 200);
   274→    }
   275→  }, [showLookupModal]);
   276→
   277→  return (
   278→    <AnimatePresence>
   279→      {showLookupModal && (
   280→        <motion.div initial={{ opacity: 0 }} animate={{ opacity: 1 }} exit={{ opacity: 0 }}
   281→          className="fixed inset-0 z-50 bg-black/50 flex items-center justify-center p-4"
   282→          onClick={() => { setShowLookupModal(false); setLookupError(''); }}>
   283→          <motion.div variants={scaleIn} initial="hidden" animate="visible" exit="exit"
   284→            onClick={e => e.stopPropagation()}
   285→            className="bg-white rounded-2xl w-full max-w-md shadow-2xl">
   286→            <div className="p-6">
   287→              <div className="flex items-center justify-between mb-6">
   288→                <div>
   289→                  <h2 className="font-semibold text-lg">My Reservation</h2>
   290→                  <p className="text-xs text-[var(--color-muted-foreground)] mt-0.5">Enter your reservation code to view your table and order</p>
   291→                </div>
   292→                <button onClick={() => { setShowLookupModal(false); setLookupError(''); }} className="p-1.5 hover:bg-[var(--color-secondary)] rounded-lg transition-colors">
   293→                  <X className="w-5 h-5" />
   294→                </button>
   295→              </div>
   296→              <div className="space-y-4">
   297→                <div className="bg-[var(--color-secondary)] rounded-xl p-4 flex items-center gap-3">
   298→                  <div className="w-10 h-10 rounded-full bg-[var(--color-accent)]/10 flex items-center justify-center flex-shrink-0">
   299→                    <Ticket className="w-5 h-5 text-[var(--color-accent)]" />
   300→                  </div>
   301→                  <div>
   302→                    <p className="text-sm font-medium">Have a reservation code?</p>
   303→                    <p className="text-xs text-[var(--color-muted-foreground)]">Enter the 6-digit code you received when you reserved</p>
   304→                  </div>
   305→                </div>
   306→                <div>
   307→                  <label className="block text-xs font-medium text-[var(--color-muted-foreground)] mb-1.5">Reservation Code</label>
   308→                  <input
   309→                    ref={inputRef}
   310→                    type="text"
   311→                    inputMode="numeric"
   312→                    pattern="\d{6}"
   313→                    value={lookupCode}
   314→                    onChange={e => handleCodeChange(e.target.value)}
   315→                    onPaste={e => {
   316→                      const pasted = e.clipboardData.getData('text').replace(/\D/g, '').slice(0, 6);
   317→                      if (pasted.length === 6) {
   318→                        setLookupCode(pasted);
   319→                        setTimeout(() => handleLookup(pasted), 100);
   320→                      } else {
   321→                        setLookupCode(pasted);
   322→                      }
   323→                    }}
   324→                    placeholder="Enter 6-digit code"
   325→                    maxLength={6}
   326→                    className="w-full px-4 py-3 border border-[var(--color-border)] rounded-xl text-lg font-mono text-center tracking-[0.5em] focus:outline-none focus:ring-2 focus:ring-[var(--color-primary)]/20 focus:border-[var(--color-primary)]"
   327→                    onKeyDown={e => { if (e.key === 'Enter') handleLookup(); }}
   328→                  />
   329→                  <div className="flex items-center justify-center gap-1 mt-2">
   330→                    {[0,1,2,3,4,5].map(i => (
   331→                      <div key={i} className={`w-2 h-2 rounded-full transition-colors ${i < lookupCode.length ? 'bg-[var(--color-primary)]' : 'bg-[var(--color-border)]'}`} />
   332→                    ))}
   333→                  </div>
   334→                </div>
   335→                <div className="flex items-center gap-2 justify-center">
   336→                  <span className="text-xs text-[var(--color-muted-foreground)] bg-[var(--color-secondary)] px-2.5 py-1 rounded-full">Test code: 777777 (Table 7)</span>
   337→                </div>
   338→                {lookupError && (
   339→                  <div className="bg-[var(--color-destructive)]/10 border border-[var(--color-destructive)]/20 rounded-xl p-3 text-center">
   340→                    <p className="text-sm text-[var(--color-destructive)]">{lookupError}</p>
   341→                  </div>
   342→                )}
   343→                <button
   344→                  onClick={() => handleLookup()}
   345→                  disabled={lookupLoading || lookupCode.length !== 6}
   346→                  className="w-full py-3 bg-[var(--color-accent)] text-white rounded-xl font-semibold text-sm hover:opacity-90 transition-opacity disabled:opacity-50 flex items-center justify-center gap-2"
   347→                >
   348→                  {lookupLoading ? <RefreshCw className="w-4 h-4 animate-spin" /> : <Key className="w-4 h-4" />}
   349→                  {lookupLoading ? 'Looking up...' : 'Find My Reservation'}
   350→                </button>
   351→              </div>
   352→            </div>
   353→          </motion.div>
   354→        </motion.div>
   355→      )}
   356→    </AnimatePresence>
   357→  );
   358→}

   363→export function CartSidebar() {
   364→  const {
   365→    cart, setCart, showCart, setShowCart,
   366→    selectedTable,
   367→    orderSubmitting,
   368→    orderSuccess,
   369→    setShowLookupModal,
   370→    submitOrder,
   371→  } = useApp();
   372→
   373→  const cartTotal = cart.reduce((sum, c) => sum + c.menuItem.price * c.quantity, 0);
   374→  const cartCount = cart.reduce((sum, c) => sum + c.quantity, 0);
   375→
   376→  const removeFromCart = (itemId: string) => {
   377→    setCart(cart.filter(c => c.menuItem.id !== itemId));
   378→  };
   379→
   380→  const updateCartQty = (itemId: string, delta: number) => {
   381→    setCart(cart.map(c => {
   382→      if (c.menuItem.id !== itemId) return c;
   383→      const newQty = c.quantity + delta;
   384→      if (newQty <= 0) return c;
   385→      return { ...c, quantity: newQty };
   386→    }).filter(c => c.quantity > 0));
   387→  };
   388→
   389→  return (
   390→    <AnimatePresence>
   391→      {showCart && (
   392→        <motion.div initial={{ opacity: 0 }} animate={{ opacity: 1 }} exit={{ opacity: 0 }}

   405→                <X className="w-5 h-5" />
   406→              </button>
   407→            </div>
   408→            <div className="flex-1 overflow-y-auto p-5">

   424→                      <div className="flex items-center gap-1.5">
   425→                        <button onClick={() => updateCartQty(item.menuItem.id, -1)} className="w-7 h-7 rounded-lg border border-[var(--color-border)] flex items-center justify-center hover:bg-[var(--color-secondary)] transition-colors">
   426→                          <Minus className="w-3 h-3" />

   440→            </div>

   463→    </AnimatePresence>

   467→// ==========================================
   468→// BILL MODAL
   469→// ==========================================
   470→export function BillModal() {
   471→  const {
   472→    showBill, setShowBill, billData, setBillData,
   473→    selectedTable, endSession,
   474→  } = useApp();
   475→
   476→  const settleBill = async () => {
   477→    if (!billData || !selectedTable) return;
   478→    try {
   479→      const billId = (billData as Record<string, unknown>).billId as string;
   480→      if (billId) {
   481→        await fetch('/api/bill', {
   482→          method: 'PATCH',
   483→          headers: { 'Content-Type': 'application/json' },
   484→          body: JSON.stringify({ billId }),
   485→        });
   486→      } else {
   487→        const orders = (billData as Record<string, unknown>).orders as { id: string }[];
   488→        for (const order of orders) {
   489→          await fetch('/api/orders', {
   490→            method: 'PATCH',
   491→            headers: { 'Content-Type': 'application/json' },
   492→            body: JSON.stringify({ orderId: order.id, status: 'billed' }),
   493→          });
   494→        }
   495→      }
   496→      setShowBill(false);
   497→      setBillData(null);
   498→      endSession();
   499→    } catch (err) {
   500→      console.error(err);
   501→    }
   502→  };
   503→
   504→  return (
   505→    <AnimatePresence>
   506→      {showBill && billData && (
   507→        <motion.div initial={{ opacity: 0 }} animate={{ opacity: 1 }} exit={{ opacity: 0 }}
   508→          className="fixed inset-0 z-50 bg-black/50 flex items-center justify-center p-4"
   509→          onClick={() => setShowBill(false)}>
   510→          <motion.div variants={scaleIn} initial="hidden" animate="visible" exit="exit"
   511→            onClick={e => e.stopPropagation()}
   512→            className="bg-white rounded-2xl w-full max-w-md shadow-2xl overflow-hidden">
   513→            <div className="bg-[var(--color-primary)] text-white p-6 text-center">
   514→              <p className="font-[var(--font-display)] text-xl font-bold">High Spirits Cafe</p>
   515→              <p className="text-xs text-white/60 mt-1">Koregaon Park, Pune</p>
   516→              <div className="mt-3 flex items-center justify-center gap-2">
   517→                <span className="px-3 py-1 bg-white/15 rounded-full text-sm font-medium">Table {(billData as Record<string, unknown>).tableNumber}</span>
   518→                <span className="px-3 py-1 bg-white/15 rounded-full text-xs">{AREA_LABELS[(billData as Record<string, unknown>).tableArea as string] || ''}</span>
   519→              </div>
   520→            </div>
   521→            <div className="p-5">
   522→              <div className="space-y-2 mb-4">
   523→                {((billData as Record<string, unknown>).items as { name: string; isVeg: boolean; quantity: number; price: number; total: number; status: string }[]).map((item, idx) => (
   524→                  <div key={idx} className="flex items-start gap-2 text-sm">
   525→                    <div className={`mt-1.5 ${item.isVeg ? 'veg-indicator' : 'nonveg-indicator'}`} style={{ flexShrink: 0 }} />
   526→                    <div className="flex-1 min-w-0">
   527→                      <span className="font-medium">{item.name}</span>
   528→                      <span className="text-[var(--color-muted-foreground)] ml-1">×{item.quantity}</span>
   529→                    </div>
   530→                    <span className="text-[var(--color-muted-foreground)] whitespace-nowrap">{fmt(item.total)}</span>
   531→                  </div>
   532→                ))}
   533→              </div>
   534→              <div className="border-t border-dashed border-[var(--color-border)] pt-3 space-y-1.5">
   535→                <div className="flex items-center justify-between text-sm">
   536→                  <span className="text-[var(--color-muted-foreground)]">Subtotal</span>
   537→                  <span>{fmt((billData as Record<string, unknown>).subtotal as number)}</span>
   538→                </div>
   539→                <div className="flex items-center justify-between text-sm">
   540→                  <span className="text-[var(--color-muted-foreground)]">GST (5%)</span>
   541→                  <span>{fmt((billData as Record<string, unknown>).gst as number)}</span>
   542→                </div>
   543→                <div className="flex items-center justify-between text-base font-bold pt-2 border-t border-[var(--color-border)]">
   544→                  <span>Total</span>
   545→                  <span>{fmt((billData as Record<string, unknown>).total as number)}</span>
   546→                </div>
   547→              </div>
   548→            </div>
   549→            <div className="p-5 pt-0 flex gap-2">
   550→              <button onClick={() => setShowBill(false)} className="flex-1 py-2.5 border border-[var(--color-border)] rounded-xl text-sm font-medium hover:bg-[var(--color-secondary)] transition-colors">
   551→                Close
   552→              </button>
   553→              <button onClick={settleBill} className="flex-1 py-2.5 bg-[var(--color-primary)] text-white rounded-xl text-sm font-medium hover:opacity-90 transition-opacity">
   554→                Settle Bill
   555→              </button>
   556→            </div>
   557→          </motion.div>
   558→        </motion.div>
   559→      )}
   560→    </AnimatePresence>
   561→  );
   562→}

   617→export function FloatingCartButton() {

   621→
   622→  if (cartCount === 0 || !selectedTable) return null;
   623→
   624→  return (
   625→    <AnimatePresence>
   626→      <motion.button
   627→        initial={{ scale: 0, y: 20 }} animate={{ scale: 1, y: 0 }} exit={{ scale: 0, y: 20 }}
   628→        transition={{ type: 'spring', damping: 20, stiffness: 300 }}

   633→        <span className="text-xs text-white/70">· {fmt(cartTotal)}</span>
