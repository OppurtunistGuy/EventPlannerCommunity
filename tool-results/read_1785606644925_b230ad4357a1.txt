   230→// ==========================================
   231→// LOOKUP RESERVATION MODAL
   232→// ==========================================
   233→export function LookupModal() {
   234→  const {
   235→    showLookupModal, setShowLookupModal,
   236→    lookupCode, setLookupCode,
   237→    lookupLoading, lookupError, setLookupError,
   238→    lookupReservationByCode,
   239→  } = useApp();
   240→
   241→  const inputRef = useRef<HTMLInputElement>(null);
   242→
   243→  const handleLookup = async (code?: string) => {
   244→    const codeToUse = code || lookupCode;
   245→    if (!codeToUse.trim() || codeToUse.trim().length !== 6) return;
   246→    const success = await lookupReservationByCode(codeToUse);
   247→    if (success) {
   248→      setShowLookupModal(false);
   249→    }
   250→  };
   251→
   252→  const handleCodeChange = (value: string) => {
   253→    // Only allow digits
   254→    const digits = value.replace(/\D/g, '').slice(0, 6);
   255→    setLookupCode(digits);
   256→    setLookupError('');
   257→    // Auto-submit when 6 digits are entered
   258→    if (digits.length === 6) {
   259→      setTimeout(() => handleLookup(digits), 100);
   260→    }
   261→  };
   262→
   263→  // Focus input when modal opens
   264→  useEffect(() => {
   265→    if (showLookupModal) {
   266→      setTimeout(() => inputRef.current?.focus(), 200);
   267→    }
   268→  }, [showLookupModal]);
   269→
   270→  return (
   271→    <AnimatePresence>
   272→      {showLookupModal && (
   273→        <motion.div initial={{ opacity: 0 }} animate={{ opacity: 1 }} exit={{ opacity: 0 }}
   274→          className="fixed inset-0 z-50 bg-black/50 flex items-center justify-center p-4"
   275→          onClick={() => { setShowLookupModal(false); setLookupError(''); }}>
   276→          <motion.div variants={scaleIn} initial="hidden" animate="visible" exit="exit"
   277→            onClick={e => e.stopPropagation()}
   278→            className="bg-white rounded-2xl w-full max-w-md shadow-2xl">
   279→            <div className="p-6">
   280→              <div className="flex items-center justify-between mb-6">
   281→                <div>
   282→                  <h2 className="font-semibold text-lg">My Reservation</h2>
   283→                  <p className="text-xs text-[var(--color-muted-foreground)] mt-0.5">Enter your reservation code to view your table and order</p>
   284→                </div>
   285→                <button onClick={() => { setShowLookupModal(false); setLookupError(''); }} className="p-1.5 hover:bg-[var(--color-secondary)] rounded-lg transition-colors">
   286→                  <X className="w-5 h-5" />
   287→                </button>
   288→              </div>
   289→              <div className="space-y-4">
   290→                <div className="bg-[var(--color-secondary)] rounded-xl p-4 flex items-center gap-3">
   291→                  <div className="w-10 h-10 rounded-full bg-[var(--color-accent)]/10 flex items-center justify-center flex-shrink-0">
   292→                    <Ticket className="w-5 h-5 text-[var(--color-accent)]" />
   293→                  </div>
   294→                  <div>
   295→                    <p className="text-sm font-medium">Have a reservation code?</p>
   296→                    <p className="text-xs text-[var(--color-muted-foreground)]">Enter the 6-digit code you received when you reserved</p>
   297→                  </div>
   298→                </div>
   299→                <div>
   300→                  <label className="block text-xs font-medium text-[var(--color-muted-foreground)] mb-1.5">Reservation Code</label>
   301→                  <input
   302→                    ref={inputRef}
   303→                    type="text"
   304→                    inputMode="numeric"
   305→                    pattern="\d{6}"
   306→                    value={lookupCode}
   307→                    onChange={e => handleCodeChange(e.target.value)}
   308→                    onPaste={e => {
   309→                      const pasted = e.clipboardData.getData('text').replace(/\D/g, '').slice(0, 6);
   310→                      if (pasted.length === 6) {
   311→                        setLookupCode(pasted);
   312→                        setTimeout(() => handleLookup(pasted), 100);
   313→                      } else {
   314→                        setLookupCode(pasted);
   315→                      }
   316→                    }}
   317→                    placeholder="Enter 6-digit code"
   318→                    maxLength={6}
   319→                    className="w-full px-4 py-3 border border-[var(--color-border)] rounded-xl text-lg font-mono text-center tracking-[0.5em] focus:outline-none focus:ring-2 focus:ring-[var(--color-primary)]/20 focus:border-[var(--color-primary)]"
   320→                    onKeyDown={e => { if (e.key === 'Enter') handleLookup(); }}
   321→                  />
   322→                  <div className="flex items-center justify-center gap-1 mt-2">
   323→                    {[0,1,2,3,4,5].map(i => (
   324→                      <div key={i} className={`w-2 h-2 rounded-full transition-colors ${i < lookupCode.length ? 'bg-[var(--color-primary)]' : 'bg-[var(--color-border)]'}`} />
   325→                    ))}
   326→                  </div>
   327→                </div>
   328→                <div className="flex items-center gap-2 justify-center">
   329→                  <span className="text-xs text-[var(--color-muted-foreground)] bg-[var(--color-secondary)] px-2.5 py-1 rounded-full">Test code: 777777 (Table 7)</span>
   330→                </div>
   331→                {lookupError && (
   332→                  <div className="bg-[var(--color-destructive)]/10 border border-[var(--color-destructive)]/20 rounded-xl p-3 text-center">
   333→                    <p className="text-sm text-[var(--color-destructive)]">{lookupError}</p>
   334→                  </div>
   335→                )}
   336→                <button
   337→                  onClick={() => handleLookup()}
   338→                  disabled={lookupLoading || lookupCode.length !== 6}
   339→                  className="w-full py-3 bg-[var(--color-accent)] text-white rounded-xl font-semibold text-sm hover:opacity-90 transition-opacity disabled:opacity-50 flex items-center justify-center gap-2"
   340→                >
   341→                  {lookupLoading ? <RefreshCw className="w-4 h-4 animate-spin" /> : <Key className="w-4 h-4" />}
   342→                  {lookupLoading ? 'Looking up...' : 'Find My Reservation'}
   343→                </button>
   344→              </div>
   345→            </div>
   346→          </motion.div>
   347→        </motion.div>
   348→      )}
   349→    </AnimatePresence>
   350→  );
   351→}

   356→export function CartSidebar() {
   357→  const {
   358→    cart, setCart, showCart, setShowCart,
   359→    selectedTable,
   360→    orderSubmitting,
   361→    orderSuccess,
   362→    setShowLookupModal,
   363→    submitOrder,
   364→  } = useApp();
   365→
   366→  const cartTotal = cart.reduce((sum, c) => sum + c.menuItem.price * c.quantity, 0);
   367→  const cartCount = cart.reduce((sum, c) => sum + c.quantity, 0);
   368→
   369→  const removeFromCart = (itemId: string) => {
   370→    setCart(cart.filter(c => c.menuItem.id !== itemId));
   371→  };
   372→
   373→  const updateCartQty = (itemId: string, delta: number) => {
   374→    setCart(cart.map(c => {
   375→      if (c.menuItem.id !== itemId) return c;
   376→      const newQty = c.quantity + delta;
   377→      if (newQty <= 0) return c;
   378→      return { ...c, quantity: newQty };
   379→    }).filter(c => c.quantity > 0));
   380→  };
   381→
   382→  return (
   383→    <AnimatePresence>
   384→      {showCart && (
   385→        <motion.div initial={{ opacity: 0 }} animate={{ opacity: 1 }} exit={{ opacity: 0 }}
   386→          className="fixed inset-0 z-50 bg-black/50" onClick={() => setShowCart(false)}>
   387→          <motion.div variants={slideInRight} initial="hidden" animate="visible" exit="exit"
   388→            onClick={e => e.stopPropagation()}
   389→            className="absolute right-0 top-0 bottom-0 w-full sm:w-[420px] bg-white shadow-2xl flex flex-col">
   390→            <div className="p-5 border-b border-[var(--color-border)] flex items-center justify-between">
   391→              <div>
   392→                <h2 className="font-semibold text-lg">Your Order</h2>
   393→                <p className="text-xs text-[var(--color-muted-foreground)] mt-0.5">
   394→                  {selectedTable ? `Table ${selectedTable.number}` : 'No table selected'}
   395→                </p>
   396→              </div>
   397→              <button onClick={() => setShowCart(false)} className="p-1.5 hover:bg-[var(--color-secondary)] rounded-lg transition-colors">
   398→                <X className="w-5 h-5" />
   399→              </button>
   400→            </div>
   401→            <div className="flex-1 overflow-y-auto p-5">
   402→              {cart.length === 0 ? (
   403→                <div className="text-center py-16 text-[var(--color-muted-foreground)]">
   404→                  <ShoppingBag className="w-10 h-10 mx-auto mb-3 opacity-30" />
   405→                  <p className="text-sm">Your cart is empty</p>
   406→                  <p className="text-xs mt-1">Add items from the menu to get started</p>
   407→                </div>
   408→              ) : (
   409→                <div className="space-y-3">
   410→                  {cart.map(item => (
   411→                    <div key={item.menuItem.id} className="flex items-start gap-3 p-3 bg-[var(--color-secondary)]/50 rounded-xl">
   412→                      <div className={`mt-1 ${item.menuItem.isVeg ? 'veg-indicator' : 'nonveg-indicator'}`} />

   414→                        <p className="text-sm font-medium leading-snug">{item.menuItem.name}</p>
   415→                        <p className="text-xs text-[var(--color-muted-foreground)] mt-0.5">{fmt(item.menuItem.price)}</p>
   416→                      </div>

   460→// ==========================================
   461→// BILL MODAL
   462→// ==========================================
   463→export function BillModal() {
   464→  const {
   465→    showBill, setShowBill, billData, setBillData,
   466→    selectedTable, endSession,
   467→  } = useApp();
   468→
   469→  const settleBill = async () => {
   470→    if (!billData || !selectedTable) return;
   471→    try {
   472→      const billId = (billData as Record<string, unknown>).billId as string;
   473→      if (billId) {
   474→        await fetch('/api/bill', {
   475→          method: 'PATCH',
   476→          headers: { 'Content-Type': 'application/json' },
   477→          body: JSON.stringify({ billId }),
   478→        });
   479→      } else {
   480→        const orders = (billData as Record<string, unknown>).orders as { id: string }[];
   481→        for (const order of orders) {
   482→          await fetch('/api/orders', {
   483→            method: 'PATCH',
   484→            headers: { 'Content-Type': 'application/json' },
   485→            body: JSON.stringify({ orderId: order.id, status: 'billed' }),
   486→          });
   487→        }
   488→      }
   489→      setShowBill(false);
   490→      setBillData(null);
   491→      endSession();
   492→    } catch (err) {
   493→      console.error(err);
   494→    }
   495→  };
   496→
   497→  return (
   498→    <AnimatePresence>
   499→      {showBill && billData && (
   500→        <motion.div initial={{ opacity: 0 }} animate={{ opacity: 1 }} exit={{ opacity: 0 }}
   501→          className="fixed inset-0 z-50 bg-black/50 flex items-center justify-center p-4"
   502→          onClick={() => setShowBill(false)}>
   503→          <motion.div variants={scaleIn} initial="hidden" animate="visible" exit="exit"
   504→            onClick={e => e.stopPropagation()}
   505→            className="bg-white rounded-2xl w-full max-w-md shadow-2xl overflow-hidden">
   506→            <div className="bg-[var(--color-primary)] text-white p-6 text-center">
   507→              <p className="font-[var(--font-display)] text-xl font-bold">High Spirits Cafe</p>
   508→              <p className="text-xs text-white/60 mt-1">Koregaon Park, Pune</p>
   509→              <div className="mt-3 flex items-center justify-center gap-2">
   510→                <span className="px-3 py-1 bg-white/15 rounded-full text-sm font-medium">Table {(billData as Record<string, unknown>).tableNumber}</span>
   511→                <span className="px-3 py-1 bg-white/15 rounded-full text-xs">{AREA_LABELS[(billData as Record<string, unknown>).tableArea as string] || ''}</span>
   512→              </div>
   513→            </div>
   514→            <div className="p-5">
   515→              <div className="space-y-2 mb-4">
   516→                {((billData as Record<string, unknown>).items as { name: string; isVeg: boolean; quantity: number; price: number; total: number; status: string }[]).map((item, idx) => (
   517→                  <div key={idx} className="flex items-start gap-2 text-sm">
   518→                    <div className={`mt-1.5 ${item.isVeg ? 'veg-indicator' : 'nonveg-indicator'}`} style={{ flexShrink: 0 }} />
   519→                    <div className="flex-1 min-w-0">
   520→                      <span className="font-medium">{item.name}</span>
   521→                      <span className="text-[var(--color-muted-foreground)] ml-1">×{item.quantity}</span>
   522→                    </div>
   523→                    <span className="text-[var(--color-muted-foreground)] whitespace-nowrap">{fmt(item.total)}</span>
   524→                  </div>
   525→                ))}
   526→              </div>
   527→              <div className="border-t border-dashed border-[var(--color-border)] pt-3 space-y-1.5">
   528→                <div className="flex items-center justify-between text-sm">
   529→                  <span className="text-[var(--color-muted-foreground)]">Subtotal</span>
   530→                  <span>{fmt((billData as Record<string, unknown>).subtotal as number)}</span>
   531→                </div>
   532→                <div className="flex items-center justify-between text-sm">
   533→                  <span className="text-[var(--color-muted-foreground)]">GST (5%)</span>
   534→                  <span>{fmt((billData as Record<string, unknown>).gst as number)}</span>
   535→                </div>
   536→                <div className="flex items-center justify-between text-base font-bold pt-2 border-t border-[var(--color-border)]">
   537→                  <span>Total</span>
   538→                  <span>{fmt((billData as Record<string, unknown>).total as number)}</span>
   539→                </div>
   540→              </div>
   541→            </div>
   542→            <div className="p-5 pt-0 flex gap-2">
   543→              <button onClick={() => setShowBill(false)} className="flex-1 py-2.5 border border-[var(--color-border)] rounded-xl text-sm font-medium hover:bg-[var(--color-secondary)] transition-colors">
   544→                Close
   545→              </button>
   546→              <button onClick={settleBill} className="flex-1 py-2.5 bg-[var(--color-primary)] text-white rounded-xl text-sm font-medium hover:opacity-90 transition-opacity">
   547→                Settle Bill
   548→              </button>
   549→            </div>
   550→          </motion.div>
   551→        </motion.div>
   552→      )}
   553→    </AnimatePresence>
   554→  );
   555→}

   615→  if (cartCount === 0 || !selectedTable) return null;
   616→
   617→  return (
   618→    <AnimatePresence>
   619→      <motion.button
   620→        initial={{ scale: 0, y: 20 }} animate={{ scale: 1, y: 0 }} exit={{ scale: 0, y: 20 }}
   621→        transition={{ type: 'spring', damping: 20, stiffness: 300 }}
   622→        onClick={() => setShowCart(true)}
