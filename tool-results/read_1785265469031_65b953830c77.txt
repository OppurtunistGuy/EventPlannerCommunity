    91→// HELPER: Happy Hour Timer
    92→// ==========================================
    93→function getHappyHourStatus() {
    94→  const now = new Date();
    95→  const ist = new Date(now.toLocaleString('en-US', { timeZone: 'Asia/Kolkata' }));
    96→  const hour = ist.getHours();
    97→  const minute = ist.getMinutes();
    98→  const currentMinutes = hour * 60 + minute;
    99→  const startMinutes = 12 * 60; // 12 PM
   100→  const endMinutes = 18 * 60; // 6 PM
   101→
   102→  if (currentMinutes >= startMinutes && currentMinutes < endMinutes) {
   103→    const remaining = endMinutes - currentMinutes;
   104→    const hoursLeft = Math.floor(remaining / 60);
   105→    const minsLeft = remaining % 60;
   106→    return {
   107→      isActive: true,
   108→      message: `Happy Hour LIVE! Ends in ${hoursLeft}h ${minsLeft}m`,
   109→      timeLeft: remaining,
   110→    };
   111→  }
   112→
   113→  if (currentMinutes < startMinutes) {
   114→    const until = startMinutes - currentMinutes;
   115→    const hoursUntil = Math.floor(until / 60);
   116→    const minsUntil = until % 60;
   117→    return {
   118→      isActive: false,
   119→      message: `Happy Hour starts in ${hoursUntil}h ${minsUntil}m`,
   120→      timeLeft: 0,
   121→    };
   122→  }
   123→
   124→  return {
   125→    isActive: false,
   126→    message: 'Happy Hour starts tomorrow at 12 PM',
   127→    timeLeft: 0,
   128→  };
   129→}
