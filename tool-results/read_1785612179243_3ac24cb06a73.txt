   240→   240→export function LookupModal() {
   241→   241→  const {
   242→   242→    showLookupModal, setShowLookupModal,
   243→   243→    lookupCode, setLookupCode,
   244→   244→    lookupLoading, lookupError, setLookupError,
   245→   245→    lookupReservationByCode,
   246→   246→  } = useApp();
   247→   247→
   248→   248→  const inputRef = useRef<HTMLInputElement>(null);
   249→   249→
   250→   250→  const handleLookup = async (code?: string) => {
   251→   251→    const codeToUse = code || lookupCode;
   252→   252→    if (!codeToUse.trim() || codeToUse.trim().length !== 6) return;
   253→   253→    const success = await lookupReservationByCode(codeToUse);
   254→   254→    if (success) {
   255→   255→      setShowLookupModal(false);
   256→   256→    }
   257→   257→  };
   258→   258→
   259→   259→  const handleCodeChange = (value: string) => {
   260→   260→    // Only allow digits
   261→   261→    const digits = value.replace(/\D/g, '').slice(0, 6);
   262→   262→    setLookupCode(digits);
   263→   263→    setLookupError('');
   264→   264→    // Auto-submit when 6 digits are entered
   265→   265→    if (digits.length === 6) {
   266→   266→      setTimeout(() => handleLookup(digits), 100);
   267→   267→    }
   268→   268→  };
   269→   269→
   270→   270→  // Focus input when modal opens
   271→   271→  useEffect(() => {
   272→   272→    if (showLookupModal) {
   273→   273→      setTimeout(() => inputRef.current?.focus(), 200);
   274→   274→    }
   275→   275→  }, [showLookupModal]);
   276→   276→
   277→   277→  return (
   278→   278→    <AnimatePresence>
   279→   279→      {showLookupModal && (
   280→   280→        <motion.div initial={{ opacity: 0 }} animate={{ opacity: 1 }} exit={{ opacity: 0 }}
   281→   281→          className="fixed inset-0 z-50 bg-black/50 flex items-center justify-center p-4"
   282→   282→          onClick={() => { setShowLookupModal(false); setLookupError(''); }}>
   283→   283→          <motion.div variants={scaleIn} initial="hidden" animate="visible" exit="exit"
   284→   284→            onClick={e => e.stopPropagation()}
   285→   285→            className="bg-white rounded-2xl w-full max-w-md shadow-2xl">
   286→   286→            <div className="p-6">
   287→   287→              <div className="flex items-center justify-between mb-6">
   288→   288→                <div>
   289→   289→                  <h2 className="font-semibold text-lg">My Reservation</h2>
   290→   290→                  <p className="text-xs text-[var(--color-muted-foreground)] mt-0.5">Enter your reservation code to view your table and order</p>
   291→   291→                </div>
   292→   292→                <button onClick={() => { setShowLookupModal(false); setLookupError(''); }} className="p-1.5 hover:bg-[var(--color-secondary)] rounded-lg transition-colors">
   293→   293→                  <X className="w-5 h-5" />
   294→   294→                </button>
   295→   295→              </div>
   296→   296→              <div className="space-y-4">
   297→   297→                <div className="bg-[var(--color-secondary)] rounded-xl p-4 flex items-center gap-3">
   298→   298→                  <div className="w-10 h-10 rounded-full bg-[var(--color-accent)]/10 flex items-center justify-center flex-shrink-0">
   299→   299→                    <Ticket className="w-5 h-5 text-[var(--color-accent)]" />
   300→   300→                  </div>
   301→   301→                  <div>
   302→   302→                    <p className="text-sm font-medium">Have a reservation code?</p>
   303→   303→                    <p className="text-xs text-[var(--color-muted-foreground)]">Enter the 6-digit code you received when you reserved</p>
   304→   304→                  </div>
   305→   305→                </div>
   306→   306→                <div>
   307→   307→                  <label className="block text-xs font-medium text-[var(--color-muted-foreground)] mb-1.5">Reservation Code</label>
   308→   308→                  <input
   309→   309→                    ref={inputRef}
   310→   310→                    type="text"
   311→   311→                    inputMode="numeric"
   312→   312→                    pattern="\d{6}"
   313→   313→                    value={lookupCode}
   314→   314→                    onChange={e => handleCodeChange(e.target.value)}
   315→   315→                    onPaste={e => {
   316→   316→                      const pasted = e.clipboardData.getData('text').replace(/\D/g, '').slice(0, 6);
   317→   317→                      if (pasted.length === 6) {
   318→   318→                        setLookupCode(pasted);
   319→   319→                        setTimeout(() => handleLookup(pasted), 100);
   320→   320→                      } else {
   321→   321→                        setLookupCode(pasted);
   322→   322→                      }
   323→   323→                    }}
   324→   324→                    placeholder="Enter 6-digit code"
   325→   325→                    maxLength={6}
   326→   326→                    className="w-full px-4 py-3 border border-[var(--color-border)] rounded-xl text-lg font-mono text-center tracking-[0.5em] focus:outline-none focus:ring-2 focus:ring-[var(--color-primary)]/20 focus:border-[var(--color-primary)]"
   327→   327→                    onKeyDown={e => { if (e.key === 'Enter') handleLookup(); }}
   328→   328→                  />
   329→   329→                  <div className="flex items-center justify-center gap-1 mt-2">
   330→   330→                    {[0,1,2,3,4,5].map(i => (
   331→   331→                      <div key={i} className={`w-2 h-2 rounded-full transition-colors ${i < lookupCode.length ? 'bg-[var(--color-primary)]' : 'bg-[var(--color-border)]'}`} />
   332→   332→                    ))}
   333→   333→                  </div>
   334→   334→                </div>
   335→   335→                <div className="flex items-center gap-2 justify-center">
   336→   336→                  <span className="text-xs text-[var(--color-muted-foreground)] bg-[var(--color-secondary)] px-2.5 py-1 rounded-full">Test code: 777777 (Table 7)</span>
   337→   337→                </div>
   338→   338→                {lookupError && (
   339→   339→                  <div className="bg-[var(--color-destructive)]/10 border border-[var(--color-destructive)]/20 rounded-xl p-3 text-center">
   340→   340→                    <p className="text-sm text-[var(--color-destructive)]">{lookupError}</p>
   341→   341→                  </div>
   342→   342→                )}
   343→   343→                <button
   344→   344→                  onClick={() => handleLookup()}
   345→   345→                  disabled={lookupLoading || lookupCode.length !== 6}
   346→   346→                  className="w-full py-3 bg-[var(--color-accent)] text-white rounded-xl font-semibold text-sm hover:opacity-90 transition-opacity disabled:opacity-50 flex items-center justify-center gap-2"
   347→   347→                >
   348→   348→                  {lookupLoading ? <RefreshCw className="w-4 h-4 animate-spin" /> : <Key className="w-4 h-4" />}
   349→   349→                  {lookupLoading ? 'Looking up...' : 'Find My Reservation'}
   350→   350→                </button>
   351→   351→              </div>
   352→   352→            </div>
   353→   353→          </motion.div>
   354→   354→        </motion.div>
   355→   355→      )}
   356→   356→    </AnimatePresence>
   357→   357→  );
   358→   358→}
