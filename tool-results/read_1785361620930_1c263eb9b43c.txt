    84→function safeISTDate(): Date {
    85→  try {
    86→    return new Date(new Date().toLocaleString('en-US', { timeZone: 'Asia/Kolkata' }));
    87→  } catch {
    88→    // Fallback: UTC+5:30 offset
    89→    const now = new Date();
    90→    return new Date(now.getTime() + (5.5 * 60 * 60 * 1000));
    91→  }
    92→}
    93→
    94→function getHappyHourEnd(): Date {
    95→  const ist = safeISTDate();
    96→  const end = new Date(ist);
    97→  end.setHours(18, 0, 0, 0);
    98→  if (ist >= end) { end.setDate(end.getDate() + 1); }
    99→  return end;
   100→}
   101→
   102→function getTimeLeft(): string {
   103→  try {
   104→    const now = safeISTDate();
   105→    const end = getHappyHourEnd();
   106→    const diff = end.getTime() - now.getTime();
   107→    if (diff <= 0) return 'Happy Hour starts tomorrow!';
   108→    const h = Math.floor(diff / 3600000);
   109→    const m = Math.floor((diff % 3600000) / 60000);
   110→    const s = Math.floor((diff % 60000) / 1000);
   111→    return `${h}h ${m}m ${s}s`;
   112→  } catch (err) {
   113→    console.error('[DIAG] getTimeLeft crashed:', err);
   114→    return '';
   115→  }
   116→}
   117→
   118→function isRestaurantOpen(): { open: boolean; closesAt: string } {
   119→  try {
   120→    const now = safeISTDate();
   121→    const hour = now.getHours();
   122→    // Open 12 PM to 1 AM (next day)
   123→    if (hour >= 12 || hour < 1) return { open: true, closesAt: '1:00 AM' };
   124→    return { open: false, closesAt: '12:00 PM' };
   125→  } catch (err) {
   126→    console.error('[DIAG] isRestaurantOpen crashed:', err);
   127→    return { open: false, closesAt: '' };
   128→  }
   129→}
