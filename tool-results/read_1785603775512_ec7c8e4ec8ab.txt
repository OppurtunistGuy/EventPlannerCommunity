   249→// LOOKUP RESERVATION MODAL
   250→// ==========================================
   251→export function LookupModal() {
   252→  const {
   253→    showLookupModal, setShowLookupModal,
   254→    lookupCode, setLookupCode,
   255→    lookupLoading, lookupError, setLookupError,
   256→    lookupReservationByCode,
   257→  } = useApp();
   258→
   259→  const inputRef = useRef<HTMLInputElement>(null);
   260→
   261→  const handleLookup = async (code?: string) => {
   262→    const codeToUse = code || lookupCode;
   263→    if (!codeToUse.trim() || codeToUse.trim().length !== 6) return;
   264→    const success = await lookupReservationByCode(codeToUse);
   265→    if (success) {
   266→      setShowLookupModal(false);
   267→    }
   268→  };
   269→
   270→  const handleCodeChange = (value: string) => {
   271→    // Only allow digits
   272→    const digits = value.replace(/\D/g, '').slice(0, 6);
   273→    setLookupCode(digits);
   274→    setLookupError('');
   275→    // Auto-submit when 6 digits are entered
   276→    if (digits.length === 6) {
   277→      setTimeout(() => handleLookup(digits), 100);
   278→    }
   279→  };
   280→
   281→  // Focus input when modal opens
   282→  useEffect(() => {
   283→    if (showLookupModal) {
   284→      setTimeout(() => inputRef.current?.focus(), 200);
   285→    }
   286→  }, [showLookupModal]);
   287→
   288→  return (
   289→    <AnimatePresence>
   290→      {showLookupModal && (
   291→        <motion.div initial={{ opacity: 0 }} animate={{ opacity: 1 }} exit={{ opacity: 0 }}
   292→          className="fixed inset-0 z-50 bg-black/50 flex items-center justify-center p-4"
   293→          onClick={() => { setShowLookupModal(false); setLookupError(''); }}>
   294→          <motion.div variants={scaleIn} initial="hidden" animate="visible" exit="exit"
   295→            onClick={e => e.stopPropagation()}
   296→            className="bg-white rounded-2xl w-full max-w-md shadow-2xl">
   297→            <div className="p-6">
   298→              <div className="flex items-center justify-between mb-6">
   299→                <div>
   300→                  <h2 className="font-semibold text-lg">My Reservation</h2>
   301→                  <p className="text-xs text-[var(--color-muted-foreground)] mt-0.5">Enter your reservation code to view your table and order</p>
   302→                </div>
   303→                <button onClick={() => { setShowLookupModal(false); setLookupError(''); }} className="p-1.5 hover:bg-[var(--color-secondary)] rounded-lg transition-colors">
   304→                  <X className="w-5 h-5" />
   305→                </button>
   306→              </div>
   307→              <div className="space-y-4">
   308→                <div className="bg-[var(--color-secondary)] rounded-xl p-4 flex items-center gap-3">
   309→                  <div className="w-10 h-10 rounded-full bg-[var(--color-accent)]/10 flex items-center justify-center flex-shrink-0">
   310→                    <Ticket className="w-5 h-5 text-[var(--color-accent)]" />
   311→                  </div>
   312→                  <div>
   313→                    <p className="text-sm font-medium">Have a reservation code?</p>
   314→                    <p className="text-xs text-[var(--color-muted-foreground)]">Enter the 6-digit code you received when you reserved</p>
   315→                  </div>
   316→                </div>
   317→                <div>
   318→                  <label className="block text-xs font-medium text-[var(--color-muted-foreground)] mb-1.5">Reservation Code</label>
   319→                  <input
   320→                    ref={inputRef}
   321→                    type="text"
   322→                    inputMode="numeric"
   323→                    pattern="\d{6}"
   324→                    value={lookupCode}
   325→                    onChange={e => handleCodeChange(e.target.value)}
   326→                    onPaste={e => {
   327→                      const pasted = e.clipboardData.getData('text').replace(/\D/g, '').slice(0, 6);
   328→                      if (pasted.length === 6) {
   329→                        setLookupCode(pasted);
   330→                        setTimeout(() => handleLookup(pasted), 100);
   331→                      } else {
   332→                        setLookupCode(pasted);
   333→                      }
   334→                    }}
   335→                    placeholder="Enter 6-digit code"
   336→                    maxLength={6}
   337→                    className="w-full px-4 py-3 border border-[var(--color-border)] rounded-xl text-lg font-mono text-center tracking-[0.5em] focus:outline-none focus:ring-2 focus:ring-[var(--color-primary)]/20 focus:border-[var(--color-primary)]"
   338→                    onKeyDown={e => { if (e.key === 'Enter') handleLookup(); }}
   339→                  />
   340→                  <div className="flex items-center justify-center gap-1 mt-2">
   341→                    {[0,1,2,3,4,5].map(i => (
   342→                      <div key={i} className={`w-2 h-2 rounded-full transition-colors ${i < lookupCode.length ? 'bg-[var(--color-primary)]' : 'bg-[var(--color-border)]'}`} />
   343→                    ))}
   344→                  </div>
   345→                </div>
   346→                <div className="flex items-center gap-2 justify-center">
   347→                  <span className="text-xs text-[var(--color-muted-foreground)] bg-[var(--color-secondary)] px-2.5 py-1 rounded-full">Test code: 777777 (Table 7)</span>
   348→                </div>
   349→                {lookupError && (
   350→                  <div className="bg-[var(--color-destructive)]/10 border border-[var(--color-destructive)]/20 rounded-xl p-3 text-center">
   351→                    <p className="text-sm text-[var(--color-destructive)]">{lookupError}</p>
   352→                  </div>
   353→                )}
   354→                <button
   355→                  onClick={() => handleLookup()}
   356→                  disabled={lookupLoading || lookupCode.length !== 6}
   357→                  className="w-full py-3 bg-[var(--color-accent)] text-white rounded-xl font-semibold text-sm hover:opacity-90 transition-opacity disabled:opacity-50 flex items-center justify-center gap-2"
   358→                >
   359→                  {lookupLoading ? <RefreshCw className="w-4 h-4 animate-spin" /> : <Key className="w-4 h-4" />}
   360→                  {lookupLoading ? 'Looking up...' : 'Find My Reservation'}
   361→                </button>
   362→              </div>
   363→            </div>
   364→          </motion.div>
   365→        </motion.div>
   366→      )}
   367→    </AnimatePresence>
   368→  );
   369→}

   387→  const removeFromCart = (itemId: string) => {
   388→    setCart(cart.filter(c => c.menuItem.id !== itemId));

   474→    </AnimatePresence>

   478→// ==========================================
   479→// BILL MODAL
   480→// ==========================================
   481→export function BillModal() {
   482→  const {
   483→    showBill, setShowBill, billData, setBillData,
   484→    selectedTable, endSession,
   485→  } = useApp();
   486→
   487→  const settleBill = async () => {
   488→    if (!billData || !selectedTable) return;
   489→    try {
   490→      const billId = (billData as Record<string, unknown>).billId as string;
   491→      if (billId) {
   492→        await fetch('/api/bill', {
   493→          method: 'PATCH',
   494→          headers: { 'Content-Type': 'application/json' },
   495→          body: JSON.stringify({ billId }),
   496→        });
   497→      } else {
   498→        const orders = (billData as Record<string, unknown>).orders as { id: string }[];
   499→        for (const order of orders) {
   500→          await fetch('/api/orders', {
   501→            method: 'PATCH',
   502→            headers: { 'Content-Type': 'application/json' },
   503→            body: JSON.stringify({ orderId: order.id, status: 'billed' }),
   504→          });
   505→        }
   506→      }
   507→      setShowBill(false);
   508→      setBillData(null);
   509→      endSession();
   510→    } catch (err) {
   511→      console.error(err);
   512→    }
   513→  };
   514→
   515→  return (
   516→    <AnimatePresence>
   517→      {showBill && billData && (
   518→        <motion.div initial={{ opacity: 0 }} animate={{ opacity: 1 }} exit={{ opacity: 0 }}
   519→          className="fixed inset-0 z-50 bg-black/50 flex items-center justify-center p-4"
   520→          onClick={() => setShowBill(false)}>
   521→          <motion.div variants={scaleIn} initial="hidden" animate="visible" exit="exit"
   522→            onClick={e => e.stopPropagation()}
   523→            className="bg-white rounded-2xl w-full max-w-md shadow-2xl overflow-hidden">
   524→            <div className="bg-[var(--color-primary)] text-white p-6 text-center">
   525→              <p className="font-[var(--font-display)] text-xl font-bold">High Spirits Cafe</p>
   526→              <p className="text-xs text-white/60 mt-1">Koregaon Park, Pune</p>
   527→              <div className="mt-3 flex items-center justify-center gap-2">
   528→                <span className="px-3 py-1 bg-white/15 rounded-full text-sm font-medium">Table {(billData as Record<string, unknown>).tableNumber}</span>
   529→                <span className="px-3 py-1 bg-white/15 rounded-full text-xs">{AREA_LABELS[(billData as Record<string, unknown>).tableArea as string] || ''}</span>
   530→              </div>
   531→            </div>
   532→            <div className="p-5">
   533→              <div className="space-y-2 mb-4">
   534→                {((billData as Record<string, unknown>).items as { name: string; isVeg: boolean; quantity: number; price: number; total: number; status: string }[]).map((item, idx) => (
   535→                  <div key={idx} className="flex items-start gap-2 text-sm">
   536→                    <div className={`mt-1.5 ${item.isVeg ? 'veg-indicator' : 'nonveg-indicator'}`} style={{ flexShrink: 0 }} />
   537→                    <div className="flex-1 min-w-0">
   538→                      <span className="font-medium">{item.name}</span>
   539→                      <span className="text-[var(--color-muted-foreground)] ml-1">×{item.quantity}</span>
   540→                    </div>
   541→                    <span className="text-[var(--color-muted-foreground)] whitespace-nowrap">{fmt(item.total)}</span>
   542→                  </div>
   543→                ))}
   544→              </div>
   545→              <div className="border-t border-dashed border-[var(--color-border)] pt-3 space-y-1.5">
   546→                <div className="flex items-center justify-between text-sm">
   547→                  <span className="text-[var(--color-muted-foreground)]">Subtotal</span>
   548→                  <span>{fmt((billData as Record<string, unknown>).subtotal as number)}</span>
   549→                </div>
   550→                <div className="flex items-center justify-between text-sm">
   551→                  <span className="text-[var(--color-muted-foreground)]">GST (5%)</span>
   552→                  <span>{fmt((billData as Record<string, unknown>).gst as number)}</span>
   553→                </div>
   554→                <div className="flex items-center justify-between text-base font-bold pt-2 border-t border-[var(--color-border)]">
   555→                  <span>Total</span>
   556→                  <span>{fmt((billData as Record<string, unknown>).total as number)}</span>
   557→                </div>
   558→              </div>
   559→            </div>
   560→            <div className="p-5 pt-0 flex gap-2">
   561→              <button onClick={() => setShowBill(false)} className="flex-1 py-2.5 border border-[var(--color-border)] rounded-xl text-sm font-medium hover:bg-[var(--color-secondary)] transition-colors">
   562→                Close
   563→              </button>
   564→              <button onClick={settleBill} className="flex-1 py-2.5 bg-[var(--color-primary)] text-white rounded-xl text-sm font-medium hover:opacity-90 transition-opacity">
   565→                Settle Bill
   566→              </button>
   567→            </div>
   568→          </motion.div>
   569→        </motion.div>
   570→      )}
   571→    </AnimatePresence>
   572→  );
   573→}

   628→export function FloatingCartButton() {

   634→

   636→    <AnimatePresence>
   637→      <motion.button
   638→        initial={{ scale: 0, y: 20 }} animate={{ scale: 1, y: 0 }} exit={{ scale: 0, y: 20 }}
