    62→function getHappyHourStatus() {
    63→  const now = new Date();
    64→  const ist = new Date(now.toLocaleString('en-US', { timeZone: 'Asia/Kolkata' }));
    65→  const m = ist.getHours() * 60 + ist.getMinutes();
    66→  if (m >= 720 && m < 1080) {
    67→    const r = 1080 - m;
    68→    return { active: true, msg: `Happy Hour is ON right now! Ends in ${Math.floor(r/60)}h ${r%60}m` };
    69→  }
    70→  return m < 720
    71→    ? { active: false, msg: 'Happy Hour starts today at 12 PM' }
    72→    : { active: false, msg: 'Happy Hour tomorrow from 12 PM' };
    73→}
