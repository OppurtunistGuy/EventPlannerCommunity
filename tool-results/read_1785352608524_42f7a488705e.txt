    84→function getHappyHourEnd(): Date {
    85→  const now = new Date();
    86→  const ist = new Date(now.toLocaleString('en-US', { timeZone: 'Asia/Kolkata' }));
    87→  const end = new Date(ist);
    88→  end.setHours(18, 0, 0, 0);
    89→  if (ist >= end) { end.setDate(end.getDate() + 1); }
    90→  return end;
    91→}
    92→
    93→function getTimeLeft(): string {
    94→  const now = new Date(new Date().toLocaleString('en-US', { timeZone: 'Asia/Kolkata' }));
    95→  const end = getHappyHourEnd();
    96→  const diff = end.getTime() - now.getTime();
    97→  if (diff <= 0) return 'Happy Hour starts tomorrow!';
    98→  const h = Math.floor(diff / 3600000);
    99→  const m = Math.floor((diff % 3600000) / 60000);
   100→  const s = Math.floor((diff % 60000) / 1000);
   101→  return `${h}h ${m}m ${s}s`;
   102→}
