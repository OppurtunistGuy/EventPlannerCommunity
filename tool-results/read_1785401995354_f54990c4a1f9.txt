    88→function safeISTDate(): Date {
    89→  try {
    90→    return new Date(new Date().toLocaleString('en-US', { timeZone: 'Asia/Kolkata' }));
    91→  } catch {
    92→    // Fallback: UTC+5:30 offset
    93→    const now = new Date();
    94→    return new Date(now.getTime() + (5.5 * 60 * 60 * 1000));
    95→  }
    96→}
    97→
    98→function getHappyHourEnd(): Date {
    99→  const ist = safeISTDate();
   100→  const end = new Date(ist);
   101→  end.setHours(18, 0, 0, 0);
   102→  if (ist >= end) { end.setDate(end.getDate() + 1); }
   103→  return end;
   104→}
   105→
   106→function getTimeLeft(): string {
   107→  try {
   108→    const now = safeISTDate();
   109→    const end = getHappyHourEnd();
   110→    const diff = end.getTime() - now.getTime();
   111→    if (diff <= 0) return 'Happy Hour starts tomorrow!';
   112→    const h = Math.floor(diff / 3600000);
   113→    const m = Math.floor((diff % 3600000) / 60000);
   114→    const s = Math.floor((diff % 60000) / 1000);
   115→    return `${h}h ${m}m ${s}s`;
   116→  } catch (err) {
   117→    console.error('[DIAG] getTimeLeft crashed:', err);
   118→    return '';
   119→  }
   120→}
   121→
   122→function isRestaurantOpen(): { open: boolean; closesAt: string } {
   123→  try {
   124→    const now = safeISTDate();
   125→    const hour = now.getHours();
   126→    // Open 12 PM to 1 AM (next day)
   127→    if (hour >= 12 || hour < 1) return { open: true, closesAt: '1:00 AM' };
   128→    return { open: false, closesAt: '12:00 PM' };
   129→  } catch (err) {
   130→    console.error('[DIAG] isRestaurantOpen crashed:', err);
   131→    return { open: false, closesAt: '' };
   132→  }
   133→}
