   306→export function CartSidebar() {
   307→  const {
   308→    cart, setCart, showCart, setShowCart,
   309→    selectedTable,
   310→    orderSubmitting,
   311→    orderSuccess,
   312→    setShowLookupModal,
   313→    submitOrder,
   314→  } = useApp();
   315→
   316→  const cartTotal = cart.reduce((sum, c) => sum + c.menuItem.price * c.quantity, 0);
   317→  const cartCount = cart.reduce((sum, c) => sum + c.quantity, 0);
   318→
   319→  const removeFromCart = (itemId: string) => {
   320→    setCart(cart.filter(c => c.menuItem.id !== itemId));
   321→  };
   322→
   323→  const updateCartQty = (itemId: string, delta: number) => {
   324→    setCart(cart.map(c => {
   325→      if (c.menuItem.id !== itemId) return c;
   326→      const newQty = c.quantity + delta;
   327→      if (newQty <= 0) return c;
   328→      return { ...c, quantity: newQty };
   329→    }).filter(c => c.quantity > 0));
   330→  };
   331→
   332→  return (
   333→    <AnimatePresence>
   334→      {showCart && (
   335→        <motion.div initial={{ opacity: 0 }} animate={{ opacity: 1 }} exit={{ opacity: 0 }}
   336→          className="fixed inset-0 z-50 bg-black/50" onClick={() => setShowCart(false)}>
   337→          <motion.div variants={slideInRight} initial="hidden" animate="visible" exit="exit"
   338→            onClick={e => e.stopPropagation()}
   339→            className="absolute right-0 top-0 bottom-0 w-full sm:w-[420px] bg-white shadow-2xl flex flex-col">
   340→            <div className="p-5 border-b border-[var(--color-border)] flex items-center justify-between">
   341→              <div>
   342→                <h2 className="font-semibold text-lg">Your Order</h2>
   343→                <p className="text-xs text-[var(--color-muted-foreground)] mt-0.5">
   344→                  {selectedTable ? `Table ${selectedTable.number}` : 'No table selected'}
   345→                </p>
   346→              </div>
   347→              <button onClick={() => setShowCart(false)} className="p-1.5 hover:bg-[var(--color-secondary)] rounded-lg transition-colors">
   348→                <X className="w-5 h-5" />
   349→              </button>
   350→            </div>
   351→            <div className="flex-1 overflow-y-auto p-5">
   352→              {cart.length === 0 ? (
   353→                <div className="text-center py-16 text-[var(--color-muted-foreground)]">
   354→                  <ShoppingBag className="w-10 h-10 mx-auto mb-3 opacity-30" />
   355→                  <p className="text-sm">Your cart is empty</p>
   356→                  <p className="text-xs mt-1">Add items from the menu to get started</p>
   357→                </div>
   358→              ) : (
   359→                <div className="space-y-3">
   360→                  {cart.map(item => (
   361→                    <div key={item.menuItem.id} className="flex items-start gap-3 p-3 bg-[var(--color-secondary)]/50 rounded-xl">
   362→                      <div className={`mt-1 ${item.menuItem.isVeg ? 'veg-indicator' : 'nonveg-indicator'}`} />
   363→                      <div className="flex-1 min-w-0">
   364→                        <p className="text-sm font-medium leading-snug">{item.menuItem.name}</p>
   365→                        <p className="text-xs text-[var(--color-muted-foreground)] mt-0.5">{fmt(item.menuItem.price)}</p>
   366→                      </div>
   367→                      <div className="flex items-center gap-1.5">
   368→                        <button onClick={() => updateCartQty(item.menuItem.id, -1)} className="w-7 h-7 rounded-lg border border-[var(--color-border)] flex items-center justify-center hover:bg-[var(--color-secondary)] transition-colors">
   369→                          <Minus className="w-3 h-3" />
   370→                        </button>
   371→                        <span className="w-6 text-center text-sm font-semibold">{item.quantity}</span>
   372→                        <button onClick={() => updateCartQty(item.menuItem.id, 1)} className="w-7 h-7 rounded-lg border border-[var(--color-border)] flex items-center justify-center hover:bg-[var(--color-secondary)] transition-colors">
   373→                          <Plus className="w-3 h-3" />
   374→                        </button>
   375→                        <button onClick={() => removeFromCart(item.menuItem.id)} className="w-7 h-7 rounded-lg flex items-center justify-center text-[var(--color-destructive)] hover:bg-red-50 transition-colors ml-1">
   376→                          <Trash2 className="w-3.5 h-3.5" />
   377→                        </button>
   378→                      </div>
   379→                    </div>
   380→                  ))}
   381→                </div>
   382→              )}
   383→            </div>
   384→            {cart.length > 0 && (
   385→              <div className="border-t border-[var(--color-border)] p-5">
   386→                <div className="flex items-center justify-between mb-4">
   387→                  <span className="text-sm text-[var(--color-muted-foreground)]">Subtotal</span>
   388→                  <span className="text-lg font-bold">{fmt(cartTotal)}</span>
   389→                </div>
   390→                {!selectedTable ? (
   391→                  <button onClick={() => { setShowCart(false); setShowLookupModal(true); }} className="w-full py-3 bg-[var(--color-accent)] text-white rounded-xl font-semibold text-sm hover:opacity-90 transition-opacity flex items-center justify-center gap-2">
   392→                    <Key className="w-4 h-4" /> Enter Reservation Code
   393→                  </button>
   394→                ) : (
   395→                  <button onClick={submitOrder} disabled={orderSubmitting}
   396→                    className="w-full py-3 bg-[var(--color-primary)] text-white rounded-xl font-semibold text-sm hover:opacity-90 transition-opacity flex items-center justify-center gap-2 disabled:opacity-50">
   397→                    {orderSubmitting ? <RefreshCw className="w-4 h-4 animate-spin" /> : <Send className="w-4 h-4" />}
   398→                    {orderSubmitting ? 'Placing Order...' : 'Place Order'}
   399→                  </button>
   400→                )}
   401→              </div>
   402→            )}
   403→          </motion.div>
   404→        </motion.div>
   405→      )}
   406→    </AnimatePresence>
   407→  );
   408→}

   413→export function BillModal() {
   414→  const {
   415→    showBill, setShowBill, billData, setBillData,
   416→    selectedTable, endSession,
   417→  } = useApp();
   418→
   419→  const settleBill = async () => {
   420→    if (!billData || !selectedTable) return;
   421→    try {
   422→      const billId = (billData as Record<string, unknown>).billId as string;
   423→      if (billId) {
   424→        await fetch('/api/bill', {
   425→          method: 'PATCH',
   426→          headers: { 'Content-Type': 'application/json' },
   427→          body: JSON.stringify({ billId }),
   428→        });
   429→      } else {
   430→        const orders = (billData as Record<string, unknown>).orders as { id: string }[];
   431→        for (const order of orders) {
   432→          await fetch('/api/orders', {
   433→            method: 'PATCH',
   434→            headers: { 'Content-Type': 'application/json' },
   435→            body: JSON.stringify({ orderId: order.id, status: 'billed' }),
   436→          });
   437→        }
   438→      }
   439→      setShowBill(false);
   440→      setBillData(null);
   441→      endSession();
   442→    } catch (err) {
   443→      console.error(err);
   444→    }
   445→  };
   446→
   447→  return (
   448→    <AnimatePresence>
   449→      {showBill && billData && (
   450→        <motion.div initial={{ opacity: 0 }} animate={{ opacity: 1 }} exit={{ opacity: 0 }}
   451→          className="fixed inset-0 z-50 bg-black/50 flex items-center justify-center p-4"
   452→          onClick={() => setShowBill(false)}>
   453→          <motion.div variants={scaleIn} initial="hidden" animate="visible" exit="exit"
   454→            onClick={e => e.stopPropagation()}
   455→            className="bg-white rounded-2xl w-full max-w-md shadow-2xl overflow-hidden">
   456→            <div className="bg-[var(--color-primary)] text-white p-6 text-center">
   457→              <p className="font-[var(--font-display)] text-xl font-bold">High Spirits Cafe</p>
   458→              <p className="text-xs text-white/60 mt-1">Koregaon Park, Pune</p>
   459→              <div className="mt-3 flex items-center justify-center gap-2">
   460→                <span className="px-3 py-1 bg-white/15 rounded-full text-sm font-medium">Table {(billData as Record<string, unknown>).tableNumber}</span>
   461→                <span className="px-3 py-1 bg-white/15 rounded-full text-xs">{AREA_LABELS[(billData as Record<string, unknown>).tableArea as string] || ''}</span>
   462→              </div>
   463→            </div>
   464→            <div className="p-5">
   465→              <div className="space-y-2 mb-4">
   466→                {((billData as Record<string, unknown>).items as { name: string; isVeg: boolean; quantity: number; price: number; total: number; status: string }[]).map((item, idx) => (
   467→                  <div key={idx} className="flex items-start gap-2 text-sm">
   468→                    <div className={`mt-1.5 ${item.isVeg ? 'veg-indicator' : 'nonveg-indicator'}`} style={{ flexShrink: 0 }} />
   469→                    <div className="flex-1 min-w-0">
   470→                      <span className="font-medium">{item.name}</span>
   471→                      <span className="text-[var(--color-muted-foreground)] ml-1">×{item.quantity}</span>
   472→                    </div>
   473→                    <span className="text-[var(--color-muted-foreground)] whitespace-nowrap">{fmt(item.total)}</span>
   474→                  </div>
   475→                ))}
   476→              </div>
   477→              <div className="border-t border-dashed border-[var(--color-border)] pt-3 space-y-1.5">
   478→                <div className="flex items-center justify-between text-sm">
   479→                  <span className="text-[var(--color-muted-foreground)]">Subtotal</span>
   480→                  <span>{fmt((billData as Record<string, unknown>).subtotal as number)}</span>
   481→                </div>
   482→                <div className="flex items-center justify-between text-sm">
   483→                  <span className="text-[var(--color-muted-foreground)]">GST (5%)</span>
   484→                  <span>{fmt((billData as Record<string, unknown>).gst as number)}</span>
   485→                </div>
   486→                <div className="flex items-center justify-between text-base font-bold pt-2 border-t border-[var(--color-border)]">
   487→                  <span>Total</span>
   488→                  <span>{fmt((billData as Record<string, unknown>).total as number)}</span>
   489→                </div>
   490→              </div>
   491→            </div>
   492→            <div className="p-5 pt-0 flex gap-2">
   493→              <button onClick={() => setShowBill(false)} className="flex-1 py-2.5 border border-[var(--color-border)] rounded-xl text-sm font-medium hover:bg-[var(--color-secondary)] transition-colors">
   494→                Close
   495→              </button>
   496→              <button onClick={settleBill} className="flex-1 py-2.5 bg-[var(--color-primary)] text-white rounded-xl text-sm font-medium hover:opacity-90 transition-opacity">
   497→                Settle Bill
   498→              </button>
   499→            </div>
   500→          </motion.div>
   501→        </motion.div>
   502→      )}
   503→    </AnimatePresence>
   504→  );
   505→}

   558→// FLOATING CART BUTTON
   559→// ==========================================
   560→export function FloatingCartButton() {
   561→  const { cart, selectedTable, setShowCart } = useApp();
   562→  const cartCount = cart.reduce((sum, c) => sum + c.quantity, 0);
   563→  const cartTotal = cart.reduce((sum, c) => sum + c.menuItem.price * c.quantity, 0);
   564→
   565→  if (cartCount === 0 || !selectedTable) return null;
   566→
   567→  return (
   568→    <AnimatePresence>
   569→      <motion.button
   570→        initial={{ scale: 0, y: 20 }} animate={{ scale: 1, y: 0 }} exit={{ scale: 0, y: 20 }}
   571→        transition={{ type: 'spring', damping: 20, stiffness: 300 }}
   572→        onClick={() => setShowCart(true)}
   573→        className="fixed bottom-6 right-6 z-40 bg-[var(--color-primary)] text-white px-5 py-3.5 rounded-2xl shadow-lg hover:shadow-xl transition-shadow flex items-center gap-3">
   574→        <ShoppingBag className="w-5 h-5" />
   575→        <span className="font-semibold text-sm">{cartCount} items</span>
   576→        <span className="text-xs text-white/70">· {fmt(cartTotal)}</span>
   577→      </motion.button>
   578→    </AnimatePresence>
   579→  );
   580→}
