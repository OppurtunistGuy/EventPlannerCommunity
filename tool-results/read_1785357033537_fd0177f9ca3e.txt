    84→    84→function getHappyHourEnd(): Date {
    85→    85→  const now = new Date();
    86→    86→  const ist = new Date(now.toLocaleString('en-US', { timeZone: 'Asia/Kolkata' }));
    87→    87→  const end = new Date(ist);
    88→    88→  end.setHours(18, 0, 0, 0);
    89→    89→  if (ist >= end) { end.setDate(end.getDate() + 1); }
    90→    90→  return end;
    91→    91→}
    92→    92→
    93→    93→function getTimeLeft(): string {
    94→    94→  const now = new Date(new Date().toLocaleString('en-US', { timeZone: 'Asia/Kolkata' }));
    95→    95→  const end = getHappyHourEnd();
    96→    96→  const diff = end.getTime() - now.getTime();
    97→    97→  if (diff <= 0) return 'Happy Hour starts tomorrow!';
    98→    98→  const h = Math.floor(diff / 3600000);
    99→    99→  const m = Math.floor((diff % 3600000) / 60000);
   100→   100→  const s = Math.floor((diff % 60000) / 1000);
   101→   101→  return `${h}h ${m}m ${s}s`;
   102→   102→}
   103→   103→
   104→   104→function isRestaurantOpen(): { open: boolean; closesAt: string } {
   105→   105→  const now = new Date(new Date().toLocaleString('en-US', { timeZone: 'Asia/Kolkata' }));
   106→   106→  const hour = now.getHours();
   107→   107→  // Open 12 PM to 1 AM (next day)
   108→   108→  if (hour >= 12 || hour < 1) return { open: true, closesAt: '1:00 AM' };
   109→   109→  return { open: false, closesAt: '12:00 PM' };
   110→   110→}
