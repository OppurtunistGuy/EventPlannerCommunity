   207→// LOOKUP RESERVATION MODAL
   208→// ==========================================
   209→export function LookupModal() {
   210→  const {
   211→    showLookupModal, setShowLookupModal,
   212→    lookupCode, setLookupCode,
   213→    lookupLoading, setLookupLoading,
   214→    lookupError, setLookupError,
   215→    setActiveReservation, setSelectedTable,
   216→  } = useApp();
   217→
   218→  const lookupReservation = async () => {
   219→    if (!lookupCode.trim()) return;
   220→    setLookupLoading(true);
   221→    setLookupError('');
   222→    try {
   223→      const res = await fetch(`/api/reservations/lookup?code=${lookupCode.trim()}`);
   224→      if (!res.ok) {
   225→        const errData = await res.json().catch(() => ({ error: 'Reservation not found' }));
   226→        setLookupError(errData.error || 'Reservation not found');
   227→        setActiveReservation(null);
   228→        return;
   229→      }
   230→      const data = await res.json();
   231→      setActiveReservation(data);
   232→      setSelectedTable(data.table);
   233→      setLookupCode('');
   234→      setShowLookupModal(false);
   235→    } catch (err) {
   236→      setLookupError('Failed to look up reservation');
   237→      setActiveReservation(null);
   238→    } finally {
   239→      setLookupLoading(false);
   240→    }
   241→  };
   242→
   243→  return (
   244→    <AnimatePresence>
   245→      {showLookupModal && (
   246→        <motion.div initial={{ opacity: 0 }} animate={{ opacity: 1 }} exit={{ opacity: 0 }}
   247→          className="fixed inset-0 z-50 bg-black/50 flex items-center justify-center p-4"
   248→          onClick={() => { setShowLookupModal(false); setLookupError(''); }}>
   249→          <motion.div variants={scaleIn} initial="hidden" animate="visible" exit="exit"
   250→            onClick={e => e.stopPropagation()}
   251→            className="bg-white rounded-2xl w-full max-w-md shadow-2xl">
   252→            <div className="p-6">
   253→              <div className="flex items-center justify-between mb-6">
   254→                <div>
   255→                  <h2 className="font-semibold text-lg">My Reservation</h2>
   256→                  <p className="text-xs text-[var(--color-muted-foreground)] mt-0.5">Enter your reservation code to view your table and order</p>
   257→                </div>
   258→                <button onClick={() => { setShowLookupModal(false); setLookupError(''); }} className="p-1.5 hover:bg-[var(--color-secondary)] rounded-lg transition-colors">
   259→                  <X className="w-5 h-5" />
   260→                </button>
   261→              </div>
   262→              <div className="space-y-4">
   263→                <div className="bg-[var(--color-secondary)] rounded-xl p-4 flex items-center gap-3">
   264→                  <div className="w-10 h-10 rounded-full bg-[var(--color-accent)]/10 flex items-center justify-center flex-shrink-0">
   265→                    <Ticket className="w-5 h-5 text-[var(--color-accent)]" />
   266→                  </div>
   267→                  <div>
   268→                    <p className="text-sm font-medium">Have a reservation code?</p>
   269→                    <p className="text-xs text-[var(--color-muted-foreground)]">Enter the 6-digit code you received when you reserved</p>
   270→                  </div>
   271→                </div>
   272→                <div>
   273→                  <label className="block text-xs font-medium text-[var(--color-muted-foreground)] mb-1.5">Reservation Code</label>
   274→                  <input
   275→                    type="text"
   276→                    value={lookupCode}
   277→                    onChange={e => { setLookupCode(e.target.value); setLookupError(''); }}
   278→                    placeholder="Enter 6-digit code"
   279→                    maxLength={6}
   280→                    className="w-full px-4 py-3 border border-[var(--color-border)] rounded-xl text-lg font-mono text-center tracking-[0.5em] focus:outline-none focus:ring-2 focus:ring-[var(--color-primary)]/20 focus:border-[var(--color-primary)]"
   281→                    onKeyDown={e => { if (e.key === 'Enter') lookupReservation(); }}
   282→                  />
   283→                </div>
   284→                <div className="flex items-center gap-2 justify-center">
   285→                  <span className="text-xs text-[var(--color-muted-foreground)] bg-[var(--color-secondary)] px-2.5 py-1 rounded-full">Test code: 777777</span>
   286→                </div>
   287→                {lookupError && (
   288→                  <p className="text-sm text-[var(--color-destructive)] text-center">{lookupError}</p>
   289→                )}
   290→                <button
   291→                  onClick={lookupReservation}
   292→                  disabled={lookupLoading || !lookupCode.trim()}
   293→                  className="w-full py-3 bg-[var(--color-accent)] text-white rounded-xl font-semibold text-sm hover:opacity-90 transition-opacity disabled:opacity-50 flex items-center justify-center gap-2"
   294→                >
   295→                  {lookupLoading ? <RefreshCw className="w-4 h-4 animate-spin" /> : <Key className="w-4 h-4" />}
   296→                  {lookupLoading ? 'Looking up...' : 'Find My Reservation'}
   297→                </button>
   298→              </div>
   299→            </div>
   300→          </motion.div>
   301→        </motion.div>
   302→      )}
   303→    </AnimatePresence>
   304→  );
   305→}

   310→export function CartSidebar() {
   311→  const {
   312→    cart, setCart, showCart, setShowCart,
   313→    selectedTable, setSelectedTable,
   314→    orderSubmitting, setOrderSubmitting,
   315→    orderSuccess, setOrderSuccess,
   316→    setShowLookupModal,
   317→  } = useApp();
   318→
   319→  const cartTotal = cart.reduce((sum, c) => sum + c.menuItem.price * c.quantity, 0);
   320→  const cartCount = cart.reduce((sum, c) => sum + c.quantity, 0);
   321→
   322→  const removeFromCart = (itemId: string) => {
   323→    setCart(cart.filter(c => c.menuItem.id !== itemId));
   324→  };
   325→
   326→  const updateCartQty = (itemId: string, delta: number) => {
   327→    setCart(cart.map(c => {
   328→      if (c.menuItem.id !== itemId) return c;
   329→      const newQty = c.quantity + delta;
   330→      if (newQty <= 0) return c;
   331→      return { ...c, quantity: newQty };
   332→    }).filter(c => c.quantity > 0));
   333→  };
   334→
   335→  const submitOrder = async () => {
   336→    if (!selectedTable || cart.length === 0) return;
   337→    setOrderSubmitting(true);
   338→    try {
   339→      const res = await fetch('/api/orders', {
   340→        method: 'POST',
   341→        headers: { 'Content-Type': 'application/json' },
   342→        body: JSON.stringify({
   343→          tableId: selectedTable.id,
   344→          type: 'dine-in',
   345→          items: cart.map(c => ({ menuItemId: c.menuItem.id, quantity: c.quantity })),
   346→        }),
   347→      });
   348→      if (!res.ok) throw new Error('Order failed');
   349→      setCart([]);
   350→      setShowCart(false);
   351→      setOrderSuccess(true);
   352→      setTimeout(() => setOrderSuccess(false), 4000);
   353→    } catch (err) {
   354→      console.error('submitOrder error:', err);
   355→    } finally {
   356→      setOrderSubmitting(false);
   357→    }
   358→  };
   359→
   360→  return (
   361→    <AnimatePresence>
   362→      {showCart && (
   363→        <motion.div initial={{ opacity: 0 }} animate={{ opacity: 1 }} exit={{ opacity: 0 }}
   364→          className="fixed inset-0 z-50 bg-black/50" onClick={() => setShowCart(false)}>
   365→          <motion.div variants={slideInRight} initial="hidden" animate="visible" exit="exit"
   366→            onClick={e => e.stopPropagation()}
   367→            className="absolute right-0 top-0 bottom-0 w-full sm:w-[420px] bg-white shadow-2xl flex flex-col">
   368→            <div className="p-5 border-b border-[var(--color-border)] flex items-center justify-between">
   369→              <div>
   370→                <h2 className="font-semibold text-lg">Your Order</h2>
   371→                <p className="text-xs text-[var(--color-muted-foreground)] mt-0.5">
   372→                  {selectedTable ? `Table ${selectedTable.number}` : 'No table selected'}
   373→                </p>
   374→              </div>
   375→              <button onClick={() => setShowCart(false)} className="p-1.5 hover:bg-[var(--color-secondary)] rounded-lg transition-colors">
   376→                <X className="w-5 h-5" />
   377→              </button>
   378→            </div>
   379→            <div className="flex-1 overflow-y-auto p-5">
   380→              {cart.length === 0 ? (
   381→                <div className="text-center py-16 text-[var(--color-muted-foreground)]">
   382→                  <ShoppingBag className="w-10 h-10 mx-auto mb-3 opacity-30" />
   383→                  <p className="text-sm">Your cart is empty</p>
   384→                  <p className="text-xs mt-1">Add items from the menu to get started</p>
   385→                </div>
   386→              ) : (
   387→                <div className="space-y-3">
   388→                  {cart.map(item => (
   389→                    <div key={item.menuItem.id} className="flex items-start gap-3 p-3 bg-[var(--color-secondary)]/50 rounded-xl">
   390→                      <div className={`mt-1 ${item.menuItem.isVeg ? 'veg-indicator' : 'nonveg-indicator'}`} />
   391→                      <div className="flex-1 min-w-0">
   392→                        <p className="text-sm font-medium leading-snug">{item.menuItem.name}</p>
   393→                        <p className="text-xs text-[var(--color-muted-foreground)] mt-0.5">{fmt(item.menuItem.price)}</p>
   394→                      </div>
   395→                      <div className="flex items-center gap-1.5">
   396→                        <button onClick={() => updateCartQty(item.menuItem.id, -1)} className="w-7 h-7 rounded-lg border border-[var(--color-border)] flex items-center justify-center hover:bg-[var(--color-secondary)] transition-colors">
   397→                          <Minus className="w-3 h-3" />
   398→                        </button>
   399→                        <span className="w-6 text-center text-sm font-semibold">{item.quantity}</span>
   400→                        <button onClick={() => updateCartQty(item.menuItem.id, 1)} className="w-7 h-7 rounded-lg border border-[var(--color-border)] flex items-center justify-center hover:bg-[var(--color-secondary)] transition-colors">
   401→                          <Plus className="w-3 h-3" />

   406→                      </div>
   407→                    </div>
   408→                  ))}
   409→                </div>
   410→              )}
   411→            </div>
   412→            {cart.length > 0 && (
   413→              <div className="border-t border-[var(--color-border)] p-5">
   414→                <div className="flex items-center justify-between mb-4">
   415→                  <span className="text-sm text-[var(--color-muted-foreground)]">Subtotal</span>
   416→                  <span className="text-lg font-bold">{fmt(cartTotal)}</span>

   540→// TABLE BAR (shows when a table is selected)
   541→// ==========================================
   542→export function TableBar() {
   543→  const {
   544→    selectedTable, setSelectedTable,
   545→    activeReservation, setActiveReservation,
   546→    billRequested, setBillRequested,
   547→    billRequesting, setBillRequesting,
   548→    setBillData, setShowBill,
   549→    setCart,
   550→  } = useApp();
   551→
   552→  const fetchBill = async (tableId: string) => {
   553→    try {
   554→      const res = await fetch(`/api/bill?tableId=${tableId}`);
   555→      if (!res.ok) throw new Error(`bill ${res.status}`);
   556→      const data = await res.json();
   557→      setBillData(data);
   558→      setShowBill(true);
   559→    } catch (err) {
   560→      console.error('fetchBill failed:', err);
   561→    }
   562→  };
   563→
   564→  const requestBill = async () => {
   565→    if (!selectedTable) return;
   566→    setBillRequesting(true);
   567→    try {
   568→      const res = await fetch('/api/bill', {
   569→        method: 'POST',
   570→        headers: { 'Content-Type': 'application/json' },
   571→        body: JSON.stringify({
   572→          tableId: selectedTable.id,
   573→          reservationId: activeReservation?.id || null,
   574→        }),
   575→      });
   576→      if (!res.ok) throw new Error('Bill request failed');
   577→      const data = await res.json();
   578→      setBillData(data);
   579→      setShowBill(true);
   580→      setBillRequested(true);
   581→    } catch (err) {
   582→      console.error('requestBill error:', err);
   583→    } finally {
   584→      setBillRequesting(false);
   585→    }
   586→  };
   587→
   588→  if (!selectedTable) return null;
   589→
   590→  return (
   591→    <AnimatePresence>
   592→      <motion.div
   593→        initial={{ height: 0, opacity: 0 }}
   594→        animate={{ height: 'auto', opacity: 1 }}
   595→        exit={{ height: 0, opacity: 0 }}
   596→        className="bg-[var(--color-primary)]/5 border-b border-[var(--color-primary)]/10 overflow-hidden"
   597→      >
   598→        <div className="max-w-6xl mx-auto px-4 sm:px-6 py-2.5 flex items-center justify-between flex-wrap gap-2">
   599→          <div className="flex items-center gap-2">
   600→            <span className="text-xs text-[var(--color-muted-foreground)]">Dining at</span>
   601→            <span className="text-sm font-semibold text-[var(--color-primary)]">Table {selectedTable.number}</span>
   602→            <span className="text-xs text-[var(--color-muted-foreground)]">· {AREA_LABELS[selectedTable.area]}</span>
   603→            {activeReservation && (
   604→              <span className="text-xs text-[var(--color-accent)] font-medium ml-1">· Code: {activeReservation.code}</span>
   605→            )}
   606→          </div>
   607→          <div className="flex items-center gap-2">
   608→            {!billRequested ? (
   609→              <button onClick={requestBill} disabled={billRequesting} className="px-3 py-1 bg-[var(--color-primary)] text-white rounded-md text-xs font-medium hover:opacity-90 transition-opacity flex items-center gap-1.5 disabled:opacity-50">
   610→                {billRequesting ? <RefreshCw className="w-3.5 h-3.5 animate-spin" /> : <FileText className="w-3.5 h-3.5" />}
   611→                {billRequesting ? 'Requesting...' : 'Bill Request'}
   612→              </button>
   613→            ) : (
   614→              <button onClick={() => fetchBill(selectedTable.id)} className="px-3 py-1 bg-[var(--color-primary)] text-white rounded-md text-xs font-medium hover:opacity-90 transition-opacity flex items-center gap-1.5">
   615→                <Receipt className="w-3.5 h-3.5" /> View bill
   616→              </button>
   617→            )}
   618→            <button onClick={() => { setSelectedTable(null); setCart([]); setActiveReservation(null); setBillRequested(false); }} className="px-3 py-1 bg-[var(--color-secondary)] rounded-md text-xs font-medium hover:bg-[var(--color-border)] transition-colors">
   619→              End Session
   620→            </button>
   621→          </div>
   622→        </div>
   623→      </motion.div>
   624→    </AnimatePresence>
   625→  );
   626→}
